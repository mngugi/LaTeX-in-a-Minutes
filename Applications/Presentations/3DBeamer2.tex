	\begin{frame}{Formalising Digital Evidence}
		
		Let
		
		\[
		E=\{e_1,e_2,\ldots,e_n\}
		\]
		
		represent the set of collected digital evidence.
		
		\vspace{0.3cm}
		
		Let
		
		\[
		A\subseteq E
		\]
		
		represent evidence that satisfies the authentication
		criteria.
		
		\vspace{0.3cm}
		
		Therefore:
		
		\[
		\boxed{A\subseteq E}
		\]
		
		\vspace{0.3cm}
		
		and
		
		\[
		e_i\in A
		\Rightarrow
		e_i\text{ satisfies the defined authentication criteria.}
		\]
		
	\end{frame}
