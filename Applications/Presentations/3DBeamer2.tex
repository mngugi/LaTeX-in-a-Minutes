	\section{Problem Statement}
	
	\begin{frame}{Problem Statement}
		
		Traditional digital forensic approaches often assume
		that evidence can be associated with identifiable
		devices, accounts, users or processes.
		
		\vspace{0.5cm}
		
		AI-assisted cybercrime introduces additional questions:
		
		\begin{itemize}
			
			\item Who actually performed the action?
			
			\item Was the action human-generated or AI-assisted?
			
			\item How can investigators establish provenance?
			
			\item Can AI-generated evidence be authenticated?
			
			\item How should attribution be established?
			
			\item How can such evidence be presented in court?
			
		\end{itemize}
		
	\end{frame}

	% ============================================================
	
	\section{Formal Reasoning}
	
	\begin{frame}{Formalising Digital Evidence}
		
		Let
		
		\[
		E = \{e_1,e_2,\ldots,e_n\}
		\]
		
		represent the set of collected digital evidence.
		
		Let
		
		\[
		A \subseteq E
		\]
		
		represent evidence that has passed the authentication
		criteria.
		
		Therefore,
		
		\[
		\boxed{A \subseteq E}
		\]
		
		and
		
		\[
		e_i \in A
		\Rightarrow
		e_i \text{ satisfies the defined authentication criteria.}
		\]
		
		\vspace{0.4cm}
		
		This creates a formal basis for reasoning about
		digital evidence.
		
	\end{frame}
